/*
  SGP41 MOX Gas Sensor — XIAO ESP32-S3
  main.cpp for PlatformIO
  -------------------------------------------------------
  Wiring:
    SGP41 VDD  →  3V3
    SGP41 GND  →  GND
    SGP41 SDA  →  D4  (GPIO5)
    SGP41 SCL  →  D5  (GPIO6)

  platformio.ini lib_deps:
    sensirion/Sensirion I2C SGP41 @ ^1.0.0
    sensirion/Sensirion Gas Index Algorithm @ ^3.2.2
*/

#include <Arduino.h>
#include <Wire.h>
#include <SensirionI2CSgp41.h>          // correct capitalisation: I2C not I2c
#include <NOxGasIndexAlgorithm.h>
#include <VOCGasIndexAlgorithm.h>

// ---------------------------------------------------------------------------
// Constants
// ---------------------------------------------------------------------------
static const uint32_t CONDITIONING_DURATION_S = 10;
static const uint32_t MEASURE_INTERVAL_MS     = 1000;

// XIAO ESP32-S3: SDA = GPIO5 (D4), SCL = GPIO6 (D5)
static const int PIN_SDA = 5;
static const int PIN_SCL = 6;

// Default compensation values — replace with live SHT4x readings if available
static const float DEFAULT_HUMIDITY_RH   = 50.0f;
static const float DEFAULT_TEMPERATURE_C = 25.0f;

// ---------------------------------------------------------------------------
// Globals
// ---------------------------------------------------------------------------
SensirionI2CSgp41    sgp41;              // correct: SensirionI2CSgp41
VOCGasIndexAlgorithm vocAlgorithm;
NOxGasIndexAlgorithm noxAlgorithm;

static bool     conditioningDone  = false;
static uint32_t conditioningStart = 0;
static uint32_t lastMeasureMs     = 0;

// ---------------------------------------------------------------------------
// Helpers
// ---------------------------------------------------------------------------

// Encode RH (%RH) and temperature (°C) into SGP41 tick format:
//   humidity    uint16 = RH  / 100 * 65535
//   temperature uint16 = (T + 45) / 175 * 65535
static uint16_t encodeHumidity(float rh) {
    return static_cast<uint16_t>(rh / 100.0f * 65535.0f);
}
static uint16_t encodeTemperature(float degC) {
    return static_cast<uint16_t>((degC + 45.0f) / 175.0f * 65535.0f);
}

// Print one row of the measurement table
static void printRow(uint32_t elapsedS,
                     uint16_t srawVoc,
                     uint16_t srawNox,
                     int32_t  vocIndex,
                     int32_t  noxIndex,
                     bool     conditioning)
{
    char buf[80];
    if (conditioning) {
        snprintf(buf, sizeof(buf),
                 "%-10lu | %-7u | %-7s | %-9s | %s",
                 (unsigned long)elapsedS, srawVoc, "-", "(cond.)", "-");
    } else {
        snprintf(buf, sizeof(buf),
                 "%-10lu | %-7u | %-7u | %-9ld | %ld",
                 (unsigned long)elapsedS, srawVoc, srawNox,
                 (long)vocIndex, (long)noxIndex);
    }
    Serial.println(buf);
}

// ---------------------------------------------------------------------------
// setup()
// ---------------------------------------------------------------------------
void setup() {
    Serial.begin(115200);
    // while (!Serial) { delay(10); }
    delay(10000);
    Serial.println(F("\n=== SGP41 + XIAO ESP32-S3 | PlatformIO Test ===\n"));

    Wire.begin(PIN_SDA, PIN_SCL);
    Wire.setClock(400000);  // 400 kHz Fast Mode

    sgp41.begin(Wire);

    // ---- Serial number — proves I2C communication works -------------------
    // getSerialNumber() returns 0 on success, non-zero error code on failure
    uint16_t serialNumber[3] = {};
    uint16_t err = sgp41.getSerialNumber(serialNumber);

    if (err != 0) {
        Serial.print(F("ERROR: could not read serial number (code 0x"));
        Serial.print(err, HEX);
        Serial.println(F(")\nCheck wiring: VDD->3V3, GND->GND, SDA->D4, SCL->D5"));
        for (;;) { delay(1000); }  // halt
    }

    Serial.print(F("SGP41 serial: 0x"));
    for (uint8_t i = 0; i < 3; i++) {
        if (serialNumber[i] < 0x1000) Serial.print('0');
        Serial.print(serialNumber[i], HEX);
    }
    Serial.println();

    // ---- Self-test ---------------------------------------------------------
    // executeSelfTest() returns 0 on success; testResult 0xD400 = all-pass
    uint16_t testResult = 0;
    err = sgp41.executeSelfTest(testResult);
    if (err != 0) {
        Serial.print(F("Self-test comm error: 0x"));
        Serial.println(err, HEX);
    } else if (testResult == 0xD400) {
        Serial.println(F("Self-test: PASSED"));
    } else {
        Serial.print(F("Self-test: FAILED (result=0x"));
        Serial.print(testResult, HEX);
        Serial.println(')');
    }

    Serial.println();
    Serial.println(F("Starting 10-second conditioning phase..."));
    Serial.println(F("(Full VOC/NOx indices appear after conditioning)\n"));
    Serial.println(F("Elapsed(s) | VOC raw | NOx raw | VOC index | NOx index"));
    Serial.println(F("-----------|---------|---------|-----------|----------"));

    conditioningStart = millis();
}

// ---------------------------------------------------------------------------
// loop()
// ---------------------------------------------------------------------------
void loop() {
    const uint32_t now = millis();
    if (now - lastMeasureMs < MEASURE_INTERVAL_MS) return;
    lastMeasureMs = now;

    const uint16_t rhTick = encodeHumidity(DEFAULT_HUMIDITY_RH);
    const uint16_t tTick  = encodeTemperature(DEFAULT_TEMPERATURE_C);

    uint16_t srawVoc = 0;
    uint16_t srawNox = 0;
    uint16_t err;

    const uint32_t elapsedMs = now - conditioningStart;
    const uint32_t elapsedS  = elapsedMs / 1000UL;

    // ---- Conditioning phase (first 10 s) -----------------------------------
    if (!conditioningDone && elapsedMs < CONDITIONING_DURATION_S * 1000UL) {
        // executeConditioning() returns 0 on success
        err = sgp41.executeConditioning(rhTick, tTick, srawVoc);
        if (err != 0) {
            Serial.print(F("Conditioning error: 0x"));
            Serial.println(err, HEX);
            return;
        }
        vocAlgorithm.process(srawVoc);  // prime the algorithm during warm-up
        printRow(elapsedS, srawVoc, 0, 0, 0, true);
        return;
    }

    // ---- Transition to normal measurement ----------------------------------
    if (!conditioningDone) {
        conditioningDone = true;
        Serial.println(F("\n- Conditioning complete. Full measurements below -\n"));
        Serial.println(F("Elapsed(s) | VOC raw | NOx raw | VOC index | NOx index"));
        Serial.println(F("-----------|---------|---------|-----------|----------"));
    }

    // ---- Normal measurement ------------------------------------------------
    // measureRawSignals() returns 0 on success
    err = sgp41.measureRawSignals(rhTick, tTick, srawVoc, srawNox);
    if (err != 0) {
        Serial.print(F("Measurement error: 0x"));
        Serial.println(err, HEX);
        return;
    }

    const int32_t vocIndex = vocAlgorithm.process(srawVoc);
    const int32_t noxIndex = noxAlgorithm.process(srawNox);

    printRow(elapsedS, srawVoc, srawNox, vocIndex, noxIndex, false);
}
